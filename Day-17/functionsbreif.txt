# ------------------------------------------------------------
# 12. *args
# ------------------------------------------------------------

def total(*numbers):
    print(numbers)


total(10, 20, 30, 40)


# *args example

def add_numbers(*numbers):
    total = 0

    for number in numbers:
        total = total + number

    return total


print(add_numbers(10, 20))
print(add_numbers(10, 20, 30))
print(add_numbers(10, 20, 30, 40))


# ------------------------------------------------------------
# 13. **kwargs
# ------------------------------------------------------------

def student_details(**details):
    print(details)


student_details(name="Abdul", age=21, course="Python")


# **kwargs example

def display_details(**details):

    for key, value in details.items():
        print(key, ":", value)


display_details(
    name="Abdul",
    age=21,
    course="Python"
)

# ------------------------------------------------------------

# ------------------------------------------------------------
# 18. RECURSIVE FUNCTION
# ------------------------------------------------------------

def factorial(n):

    if n == 1:
        return 1

    return n * factorial(n - 1)


print("Factorial:", factorial(5))


# ------------------------------------------------------------
# 19. LAMBDA FUNCTION
# ------------------------------------------------------------

square = lambda x: x * x

print("Square:", square(5))


# Lambda with two arguments

add = lambda a, b: a + b

print("Lambda Addition:", add(10, 20))


# ------------------------------------------------------------
# 20. HIGHER-ORDER FUNCTION
# ------------------------------------------------------------

def square_number(n):

    return n * n


def calculate(function, number):

    return function(number)


result = calculate(square_number, 5)

print("Higher Order Function:", result)

# ------------------------------------------------------------
# 22. MULTIPLE RETURN VALUES
# ------------------------------------------------------------

def calculate(a, b):

    addition = a + b
    subtraction = a - b
    multiplication = a * b

    return addition, subtraction, multiplication


x, y, z = calculate(10, 5)

print("Addition:", x)
print("Subtraction:", y)
print("Multiplication:", z)
